\begin{abstract}
This reproducible example compares a nominal model with a
regularized alternative under small data perturbations.
\end{abstract}

\section{Model}
For observations $A \in \mathbb{R}^{m \times n}$ and targets
$b \in \mathbb{R}^{m}$, consider
\begin{equation}
  \hat{x}
  = \arg\min_x
    \left\lVert Ax-b \right\rVert_2^2
    + \lambda \left\lVert x \right\rVert_2^2.
  \label{eq:model}
\end{equation}

\section{Result}
The regularized estimate remains stable as the noise level grows.
